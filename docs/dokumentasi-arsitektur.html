3️⃣ Kode Migrasi (struktur tabel dalam bentuk kode)

database/migrations/2026_01_01_000001_add_role_scope_and_create_units_table.php

// Menambah kolom role, uid, up3, ulp ke tabel users & membuat tabel units
Schema::table('users', function (Blueprint $table) {
    $table->string('role')->default('operator')->after('password');
    $table->string('uid')->nullable()->after('role');
    $table->string('up3')->nullable()->after('uid');
    $table->string('ulp')->nullable()->after('up3');
});

Schema::create('units', function (Blueprint $table) {
    $table->id();
    $table->string('nama');
    $table->enum('jenis', ['up3', 'ulp']);
    $table->string('kode')->nullable();
    $table->string('parent_up3')->nullable();
    $table->timestamps();
});

// Filter data berdasarkan scope role (UID > UP3 > ULP)
public function scopeData(array $data): array {
    $user = Auth::user();
    if (!$user || !$user->canViewUnitData()) {
        return [];   // super admin tidak lihat data unit
    }
    $scope = $user->dataScope();
    if ($scope['level'] === 'uid') {
        return $data;
    }
    ...
}

7️⃣ Model User — scope data & helper role (app/Models/User.php)

protected $fillable = [
    'name', 'email', 'password',
    'role', 'uid', 'up3', 'ulp',
];

// Prioritas scope: UID > UP3 > ULP
public function dataScope(): array
{
    if ($this->uid)  return ['level' => 'uid',  'value' => $this->uid];
    if ($this->up3) return ['level' => 'up3', 'value' => $this->up3];
    if ($this->ulp) return ['level' => 'ulp', 'value' => $this->ulp];
    return ['level' => 'uid', 'value' => null];
}
